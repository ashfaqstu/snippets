-- ------------------------------------------------------------------------------
-- QUESTION A1: 
-- Procedure LONGEST_SERVING_EMPLOYEE taking a REGION_NAME as input. [cite: 141]
-- Finds the employee working the longest in that region and prints their details. [cite: 142, 143]
-- ------------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE LONGEST_SERVING_EMPLOYEE (
    p_region_name IN regions.region_name%TYPE
) IS
    v_first_name employees.first_name%TYPE;
    v_last_name  employees.last_name%TYPE;
    v_job_title  jobs.job_title%TYPE;
    v_hire_date  employees.hire_date%TYPE;
    v_country    countries.country_name%TYPE;
    v_department departments.department_name%TYPE;
    v_city       locations.city%TYPE;
BEGIN
    SELECT e.first_name, e.last_name, j.job_title, e.hire_date, c.country_name, d.department_name, l.city
    INTO v_first_name, v_last_name, v_job_title, v_hire_date, v_country, v_department, v_city
    FROM employees e
    JOIN jobs j ON e.job_id = j.job_id
    JOIN departments d ON e.department_id = d.department_id
    JOIN locations l ON d.location_id = l.location_id
    JOIN countries c ON l.country_id = c.country_id
    JOIN regions r ON c.region_id = r.region_id
    WHERE r.region_name = p_region_name
    ORDER BY e.hire_date ASC
    FETCH FIRST 1 ROW ONLY;

    DBMS_OUTPUT.PUT_LINE('Longest Serving Employee in Region: ' || p_region_name);
    DBMS_OUTPUT.PUT_LINE('Name: ' || v_first_name || ' ' || v_last_name);
    DBMS_OUTPUT.PUT_LINE('Job Title: ' || v_job_title);
    DBMS_OUTPUT.PUT_LINE('Hire Date: ' || v_hire_date);
    DBMS_OUTPUT.PUT_LINE('Country: ' || v_country);
    DBMS_OUTPUT.PUT_LINE('Department: ' || v_department);
    DBMS_OUTPUT.PUT_LINE('City: ' || v_city);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No employees found for the specified region.');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
END;
/

-- ------------------------------------------------------------------------------
-- QUESTION A2: 
-- Procedure RANK_JOBS to rank jobs by number of employees (desc). [cite: 153, 154]
-- Tie-breaker: average salary (desc). [cite: 156] 
-- Prints Rank, Job Title, Employees, Avg Sal, Max Sal, Min Sal. [cite: 157, 159, 161, 162, 163, 164, 165]
-- ------------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE RANK_JOBS IS
    CURSOR c_jobs IS
        SELECT j.job_title, COUNT(e.employee_id) as num_emps, 
               NVL(AVG(e.salary), 0) as avg_sal, j.max_salary, j.min_salary
        FROM jobs j
        LEFT JOIN employees e ON j.job_id = e.job_id
        GROUP BY j.job_title, j.max_salary, j.min_salary
        ORDER BY num_emps DESC, avg_sal DESC;
    v_rank NUMBER := 1;
BEGIN
    FOR r IN c_jobs LOOP
        DBMS_OUTPUT.PUT_LINE('Rank ' || v_rank);
        DBMS_OUTPUT.PUT_LINE('Job Title: ' || r.job_title);
        DBMS_OUTPUT.PUT_LINE('Employees: ' || r.num_emps);
        DBMS_OUTPUT.PUT_LINE('Avg Salary: ' || ROUND(r.avg_sal, 2));
        DBMS_OUTPUT.PUT_LINE('Max Salary: ' || r.max_salary);
        DBMS_OUTPUT.PUT_LINE('Min Salary: ' || r.min_salary);
        DBMS_OUTPUT.PUT_LINE('---------------------------');
        v_rank := v_rank + 1;
    END LOOP;
END;
/

-- ------------------------------------------------------------------------------
-- QUESTION C1: 
-- Function IS_READY_FOR_PROMOTION. Returns 'YES' if: >= 5 yrs since hire, [cite: 6, 9, 10]
-- salary > midpoint of min/max, and manages >= 1 subordinate. Else 'NO'. [cite: 11, 12, 13]
-- ------------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION IS_READY_FOR_PROMOTION (
    p_emp_id IN employees.employee_id%TYPE
) RETURN VARCHAR2 IS
    v_hire_date employees.hire_date%TYPE;
    v_salary employees.salary%TYPE;
    v_job_id employees.job_id%TYPE;
    v_min_sal jobs.min_salary%TYPE;
    v_max_sal jobs.max_salary%TYPE;
    v_sub_count NUMBER;
    v_midpoint NUMBER;
BEGIN
    -- Get employee details
    SELECT e.hire_date, e.salary, e.job_id, j.min_salary, j.max_salary
    INTO v_hire_date, v_salary, v_job_id, v_min_sal, v_max_sal
    FROM employees e
    JOIN jobs j ON e.job_id = j.job_id
    WHERE e.employee_id = p_emp_id;

    -- Get subordinate count
    SELECT COUNT(*) INTO v_sub_count
    FROM employees
    WHERE manager_id = p_emp_id;

    v_midpoint := (v_min_sal + v_max_sal) / 2;

    -- Check conditions
    IF MONTHS_BETWEEN(SYSDATE, v_hire_date) >= 60 
       AND v_salary > v_midpoint 
       AND v_sub_count >= 1 THEN
        RETURN 'YES';
    ELSE
        RETURN 'NO';
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN 'NO'; -- Employee exists but fails conditions or doesn't exist
    WHEN OTHERS THEN
        RETURN 'ERROR: ' || SQLERRM;
END;
/

-- ------------------------------------------------------------------------------
-- QUESTION B2: 
-- Procedure LOCATION_SALARY_REPORT to compute per city: number of emps, [cite: 104, 105, 108]
-- avg salary (rounded to 2 decimals), and job title of highest-paid emp. [cite: 109, 110]
-- Rank asc by num emps, desc by avg salary. [cite: 111, 112, 113]
-- ------------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE LOCATION_SALARY_REPORT IS
    CURSOR c_locations IS
        WITH CityStats AS (
            SELECT l.city, COUNT(e.employee_id) as num_emps, ROUND(AVG(e.salary), 2) as avg_sal
            FROM locations l
            JOIN departments d ON l.location_id = d.location_id
            JOIN employees e ON d.department_id = e.department_id
            GROUP BY l.city
        ),
        HighestPaid AS (
            SELECT l.city, j.job_title,
                   ROW_NUMBER() OVER(PARTITION BY l.city ORDER BY e.salary DESC) as rn
            FROM locations l
            JOIN departments d ON l.location_id = d.location_id
            JOIN employees e ON d.department_id = e.department_id
            JOIN jobs j ON e.job_id = j.job_id
        )
        SELECT cs.city, cs.num_emps, cs.avg_sal, hp.job_title
        FROM CityStats cs
        JOIN HighestPaid hp ON cs.city = hp.city AND hp.rn = 1
        ORDER BY cs.num_emps ASC, cs.avg_sal DESC;
        
    v_rank NUMBER := 1;
BEGIN
    FOR r IN c_locations LOOP
        DBMS_OUTPUT.PUT_LINE('Rank: ' || v_rank || ' | City: ' || r.city || 
                             ' | Employees: ' || r.num_emps || 
                             ' | Avg Salary: ' || r.avg_sal || 
                             ' | Highest Paying Job: ' || r.job_title);
        v_rank := v_rank + 1;
    END LOOP;
EXCEPTION
    WHEN OTHERS THEN
         DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;
/

-- ------------------------------------------------------------------------------
-- QUESTION B1: 
-- Procedure Exchange_Employees using EMPLOYEES_COPY.
-- Swaps lowest-paid employees under two managers.
-- New Salary = current + 0.5 * (difference between original salaries).
-- ------------------------------------------------------------------------------
-- Note: Ensure you run: CREATE TABLE employees_copy AS SELECT * FROM employees;
-- ------------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE Exchange_Employees (
    p_mgr_id1 IN NUMBER,
    p_mgr_id2 IN NUMBER
) IS
    v_emp1 employees_copy%ROWTYPE;
    v_emp2 employees_copy%ROWTYPE;
    v_sal_diff NUMBER;
BEGIN
    -- Get lowest paid employee for Manager 1
    SELECT * INTO v_emp1 FROM (
        SELECT * FROM employees_copy WHERE manager_id = p_mgr_id1 ORDER BY salary ASC
    ) WHERE ROWNUM = 1;

    -- Get lowest paid employee for Manager 2
    SELECT * INTO v_emp2 FROM (
        SELECT * FROM employees_copy WHERE manager_id = p_mgr_id2 ORDER BY salary ASC
    ) WHERE ROWNUM = 1;

    v_sal_diff := ABS(v_emp1.salary - v_emp2.salary);

    -- Swap Manager and Department, Update Salary
    UPDATE employees_copy 
    SET manager_id = p_mgr_id2, 
        department_id = v_emp2.department_id,
        salary = salary + (0.5 * v_sal_diff)
    WHERE employee_id = v_emp1.employee_id;

    UPDATE employees_copy 
    SET manager_id = p_mgr_id1, 
        department_id = v_emp1.department_id,
        salary = salary + (0.5 * v_sal_diff)
    WHERE employee_id = v_emp2.employee_id;

    DBMS_OUTPUT.PUT_LINE('Exchanged ' || v_emp1.first_name || ' with ' || v_emp2.first_name);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('One or both managers have no employees.');
END;
/

-- ------------------------------------------------------------------------------
-- QUESTION A3: 
-- Trigger for LEAVES table on DELETE.
-- Logic: If employee is a manager, replacement is manager with closest sub-count.
-- If not a manager, replacement is peer with closest salary under same manager.
-- ------------------------------------------------------------------------------
CREATE TABLE Leaves (
    employee_id NUMBER,
    employee_working_instead_of_him NUMBER,
    leave_date DATE
);

CREATE OR REPLACE TRIGGER trg_employee_departure
FOR DELETE ON employees
COMPOUND TRIGGER
    -- Capture old data to avoid mutating table error
    TYPE t_emp_rec IS RECORD (id NUMBER, mgr NUMBER, sal NUMBER);
    TYPE t_emp_list IS TABLE OF t_emp_rec;
    v_list t_emp_list := t_emp_list();

    BEFORE EACH ROW IS
    BEGIN
        v_list.EXTEND;
        v_list(v_list.LAST).id := :OLD.employee_id;
        v_list(v_list.LAST).mgr := :OLD.manager_id;
        v_list(v_list.LAST).sal := :OLD.salary;
    END BEFORE EACH ROW;

    AFTER STATEMENT IS
        v_replacement NUMBER;
        v_sub_count NUMBER;
    BEGIN
        FOR i IN 1..v_list.COUNT LOOP
            -- Check if deleted employee was a manager
            SELECT COUNT(*) INTO v_sub_count FROM employees WHERE manager_id = v_list(i).id;

            IF v_sub_count > 0 THEN
                -- Find manager with closest subordinate count
                SELECT mgr_id INTO v_replacement FROM (
                    SELECT manager_id as mgr_id, COUNT(*) as cnt 
                    FROM employees GROUP BY manager_id
                    ORDER BY ABS(COUNT(*) - v_sub_count) ASC
                ) WHERE ROWNUM = 1;
            ELSE
                -- Find peer with closest salary
                SELECT employee_id INTO v_replacement FROM (
                    SELECT employee_id FROM employees 
                    WHERE manager_id = v_list(i).mgr
                    ORDER BY ABS(salary - v_list(i).sal) ASC
                ) WHERE ROWNUM = 1;
            END IF;

            INSERT INTO Leaves VALUES (v_list(i).id, v_replacement, SYSDATE);
        END LOOP;
    END AFTER STATEMENT;
END;
/

-- ------------------------------------------------------------------------------
-- QUESTION B3: 
-- Trigger for TRANSFERS table on UPDATE of department_id.
-- Replacement: Closest salary under same manager.
-- ------------------------------------------------------------------------------
CREATE TABLE Transfers (
    employee_id NUMBER,
    employee_working_instead_of_him NUMBER,
    new_department NUMBER,
    transfer_date DATE
);

CREATE OR REPLACE TRIGGER trg_dept_transfer
AFTER UPDATE OF department_id ON employees
FOR EACH ROW
DECLARE
    v_replacement NUMBER;
BEGIN
    -- Find peer with closest salary under the same manager
    SELECT employee_id INTO v_replacement FROM (
        SELECT employee_id FROM employees 
        WHERE manager_id = :OLD.manager_id 
        AND employee_id != :OLD.employee_id
        ORDER BY ABS(salary - :OLD.salary) ASC
    ) WHERE ROWNUM = 1;

    INSERT INTO Transfers VALUES (:OLD.employee_id, v_replacement, :NEW.department_id, SYSDATE);
EXCEPTION
    WHEN NO_DATA_FOUND THEN NULL; -- Handle case where no peer exists
END;
/

-- ------------------------------------------------------------------------------
-- QUESTION C3: 
-- Trigger for DEMOTIONS table on salary decrease.
-- Replacement: Closest hire_date under same manager.
-- ------------------------------------------------------------------------------
CREATE TABLE Demotions (
    employee_id NUMBER,
    employee_working_instead_of_him NUMBER,
    old_salary NUMBER,
    new_salary NUMBER,
    current_date_demoted DATE
);

CREATE OR REPLACE TRIGGER trg_salary_demotion
AFTER UPDATE OF salary ON employees
FOR EACH ROW
WHEN (NEW.salary < OLD.salary)
DECLARE
    v_replacement NUMBER;
BEGIN
    -- Find peer with closest hire date under the same manager
    SELECT employee_id INTO v_replacement FROM (
        SELECT employee_id FROM employees 
        WHERE manager_id = :OLD.manager_id 
        AND employee_id != :OLD.employee_id
        ORDER BY ABS(hire_date - :OLD.hire_date) ASC
    ) WHERE ROWNUM = 1;

    INSERT INTO Demotions VALUES (:OLD.employee_id, v_replacement, :OLD.salary, :NEW.salary, SYSDATE);
EXCEPTION
    WHEN NO_DATA_FOUND THEN NULL;
END;
